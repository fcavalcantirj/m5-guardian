# Host build of the portable Guardian layers (G03). No ESP-IDF runtime, no M5Stack.
# (G22A: one check, tune_offer_check, compiles twelve mbedtls source files read from the
# pinned ESP-IDF checkout; see GUARDIAN_MBEDTLS_DIR below.)
#
#   cmake -S firmware -B <build dir> [-DGUARDIAN_SANITIZE=ON]
#   cmake --build <build dir>
#   ctest --test-dir <build dir> --output-on-failure
#
# <build dir> belongs in the store (firmware/tools/host-check.sh does this and keeps
# the logs). Libraries: guardian_core, guardian_presenter, guardian_canvas, the
# same sources the ESP-IDF component CMakeLists register. Tests: one executable
# per concern plus the headless replay example, all linked against the real
# libraries. The board project under boards/sticks3 does not use this file.
cmake_minimum_required(VERSION 3.16)
project(guardian_host LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
endif()

option(GUARDIAN_SANITIZE "Build everything with AddressSanitizer + UndefinedBehaviorSanitizer" OFF)
if(GUARDIAN_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all -g)
  add_link_options(-fsanitize=address,undefined)
endif()

# Library policy = the one G01 was reviewed under. The stricter set is applied
# only to NEW G03 test code: the pre-existing canvas has int->size_t implicit
# conversions (6 sites) that -Wsign-conversion rejects; changing library code for
# warnings is not in G03 scope and is recorded in docs/G03-host-harness.md.
set(GUARDIAN_WARNINGS -Wall -Wextra -Wpedantic -Werror)
set(GUARDIAN_TEST_WARNINGS ${GUARDIAN_WARNINGS} -Wshadow -Wconversion -Wsign-conversion)

# G22A: a TUNE offer is a signed envelope. The signature check (adapters/tune/offer_rsa.cpp) is mbedtls on the
# board, so the host check builds the same mbedtls sources: the submodule of the ESP-IDF checkout the board
# build pins (boards/sticks3/tools/build.sh IDF_PATH_PINNED), read in place, nothing copied into this repository.
# Only the twelve files RSA PKCS#1 v1.5 + SHA-256 need, with tests/support/guardian_mbedtls_config.h, -O2 (the
# check generates its own throwaway RSA-3072 key pair). A missing checkout stops the configure: a silently
# skipped signature check would be worse than no build.
set(GUARDIAN_MBEDTLS_DIR "$ENV{HOME}/esp/esp-idf/components/mbedtls/mbedtls" CACHE PATH "mbedtls source tree (ESP-IDF v5.4 submodule) for the G22A offer signature check")
if(NOT EXISTS "${GUARDIAN_MBEDTLS_DIR}/library/rsa.c" OR NOT EXISTS "${GUARDIAN_MBEDTLS_DIR}/include/mbedtls/rsa.h")
  message(FATAL_ERROR "G22A: mbedtls sources not found at GUARDIAN_MBEDTLS_DIR=${GUARDIAN_MBEDTLS_DIR} (the ESP-IDF checkout's components/mbedtls/mbedtls). tune_offer_check cannot be built without them.")
endif()
enable_language(C)
set(GUARDIAN_MBEDTLS_SOURCES "")
foreach(file bignum bignum_core rsa rsa_alt_helpers constant_time platform_util platform md sha256 oid asn1parse asn1write)
  list(APPEND GUARDIAN_MBEDTLS_SOURCES "${GUARDIAN_MBEDTLS_DIR}/library/${file}.c")
endforeach()
add_library(guardian_host_mbedtls STATIC ${GUARDIAN_MBEDTLS_SOURCES})
target_include_directories(guardian_host_mbedtls PUBLIC "${GUARDIAN_MBEDTLS_DIR}/include" tests/support)
target_compile_definitions(guardian_host_mbedtls PUBLIC MBEDTLS_CONFIG_FILE="guardian_mbedtls_config.h")
target_compile_options(guardian_host_mbedtls PRIVATE -O2 -w)   # third-party C: optimised, its warnings are not ours
add_library(guardian_tune_offer_rsa STATIC adapters/tune/offer_rsa.cpp)
target_link_libraries(guardian_tune_offer_rsa PUBLIC guardian_tune_session guardian_host_mbedtls)
target_compile_options(guardian_tune_offer_rsa PRIVATE ${GUARDIAN_WARNINGS})
add_executable(tune_offer_check tests/tune_offer_check.cpp)
target_link_libraries(tune_offer_check PRIVATE guardian_tune_offer_rsa guardian_test_support)
target_compile_options(tune_offer_check PRIVATE ${GUARDIAN_TEST_WARNINGS} -Wno-sign-conversion)
add_test(NAME tune_offer_check COMMAND tune_offer_check)

function(guardian_add_test name)
  add_executable(${name} tests/${name}.cpp)
  target_link_libraries(${name} PRIVATE guardian_test_support ${ARGN})
  target_compile_options(${name} PRIVATE ${GUARDIAN_TEST_WARNINGS})
  add_test(NAME ${name} COMMAND ${name})
endfunction()

# Regression checks of the test runner itself. Deliberately failing child cases
# must have the expected nonzero status; they are not vacuous passing tests.
add_executable(check_framework_probe tests/check_framework_probe.cpp)
target_link_libraries(check_framework_probe PRIVATE guardian_test_support)
target_compile_options(check_framework_probe PRIVATE ${GUARDIAN_TEST_WARNINGS})
function(guardian_framework_check name expected)
  add_test(NAME ${name} COMMAND ${CMAKE_COMMAND}
    "-DPROGRAM=$<TARGET_FILE:check_framework_probe>" "-DEXPECT_EXIT=${expected}"
    "-DARGS=${ARGN}" -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/expect_process.cmake)
endfunction()
guardian_framework_check(framework_positive_control 0 one_check)
guardian_framework_check(framework_rejects_empty_case 1 empty_case)
guardian_framework_check(framework_rejects_empty_mixed 1 one_check empty_case)
guardian_framework_check(framework_rejects_unknown_mixed 3 one_check typo_case)
guardian_framework_check(framework_rejects_unknown_only 3 typo_case)
guardian_framework_check(framework_rejects_failed_assertion 1 failing_case)
# Matching output alone must never turn a nonzero child exit into success.
add_test(NAME process_guard_rejects_matching_error_exit COMMAND ${CMAKE_COMMAND}
  "-DPROGRAM=$<TARGET_FILE:check_framework_probe>" -DEXPECT_EXIT=0
  -DARGS=failing_case "-DOUTPUT_PATTERN=RESULT: FAIL"
  -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/expect_process.cmake)

# ---- optional independent decoder (G04 reviewer acceptance aid) -------------------------------
# libwifi (Apache-2.0, pinned c38dbc8 in the store, see docs/research/reuse-decisions.md) is compiled
# HOST-ONLY as a differential reference for individually vetted valid fixtures. It is never linked
# into guardian_core or any firmware target. Opt in with
#   -DGUARDIAN_LIBWIFI_DIR=$GUARDIAN_STORE/third-party/libwifi
# The default build has no third-party dependency. The oracle has known defects (G02: low-byte
# counts, 6-suite clamp, over-read without RSN capabilities, stops at a later zero-length element);
# tests/parser_oracle_libwifi.cpp compares only fixtures those defects cannot touch and pins the
# documented disagreement for the zero-length-element fixture.
set(GUARDIAN_LIBWIFI_DIR "" CACHE PATH "Pinned libwifi checkout for the host-only differential oracle (empty = off)")
if(GUARDIAN_LIBWIFI_DIR)
  if(NOT EXISTS "${GUARDIAN_LIBWIFI_DIR}/src/libwifi/parse/misc/security.c")
    message(FATAL_ERROR "GUARDIAN_LIBWIFI_DIR=${GUARDIAN_LIBWIFI_DIR} does not contain src/libwifi (pinned libwifi checkout expected)")
  endif()
  enable_language(C)
  file(GLOB_RECURSE LIBWIFI_SOURCES "${GUARDIAN_LIBWIFI_DIR}/src/libwifi/*.c")
  list(FILTER LIBWIFI_SOURCES EXCLUDE REGEX "/\\._") # macOS AppleDouble shadows on the SMB share
  add_library(libwifi_oracle STATIC ${LIBWIFI_SOURCES})
  target_include_directories(libwifi_oracle SYSTEM PUBLIC "${GUARDIAN_LIBWIFI_DIR}/src")
  set_target_properties(libwifi_oracle PROPERTIES C_STANDARD 11 C_EXTENSIONS ON)
  target_compile_options(libwifi_oracle PRIVATE -w) # third-party code, not Guardian's warning policy
  guardian_add_test(parser_oracle_libwifi guardian_core libwifi_oracle)
  target_compile_definitions(parser_oracle_libwifi PRIVATE GUARDIAN_LIBWIFI_DIR="${GUARDIAN_LIBWIFI_DIR}")
  message(STATUS "G04 libwifi oracle enabled from ${GUARDIAN_LIBWIFI_DIR}")
endif()

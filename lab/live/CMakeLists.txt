cmake_minimum_required(VERSION 3.16)
project(guardian_live LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GUARDIAN_ROOT "" CACHE PATH "Guardian repository root")
if(NOT EXISTS "${GUARDIAN_ROOT}/firmware/CMakeLists.txt")
 message(FATAL_ERROR "Set GUARDIAN_ROOT to the Guardian repository")
endif()
add_subdirectory("${GUARDIAN_ROOT}/firmware" guardian-host EXCLUDE_FROM_ALL)
add_executable(guardian_live_engine engine.cpp)
target_link_libraries(guardian_live_engine PRIVATE guardian_presenter)
target_compile_options(guardian_live_engine PRIVATE -Wall -Wextra -Werror)
if(GUARDIAN_SANITIZE)
 target_compile_options(guardian_live_engine PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
 target_link_options(guardian_live_engine PRIVATE -fsanitize=address,undefined)
endif()
